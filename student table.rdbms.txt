DECLARE

rno int;
sname char(10);
per float;
class varchar(5);

BEGIN

rno:=&rno;
sname:=&sname;
per:=&per;
class:=&class;

if per>=40 and per<=50 Then
             class:='pass';
else if per>50 and per<=60 Then
              class:='C';
else if per>60 and per<=70 Then
              class:='B';
else if per>70 and per<=80 Then
              class:='A';
else if per>80 and per<=100 Then
              class:='O';
else
        class:='Fail';
END if; END if;END if; END if;END if;
